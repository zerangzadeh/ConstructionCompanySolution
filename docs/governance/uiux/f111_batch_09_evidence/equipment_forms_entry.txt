class _StyledForm(JalaliDateMixin, forms.Form):
    def __init__(self, *args: object, **kwargs: object) -> None:
        super().__init__(*args, **kwargs)
        field_class = (
            "block w-full rounded-lg border border-slate-300 bg-white px-3 py-2.5 "
            "text-slate-900 outline-none focus:border-blue-500 focus:ring-2 "
            "focus:ring-blue-100"
        )
        for field in self.fields.values():
            field.widget.attrs.setdefault("class", field_class)

class EquipmentAssignmentForm(_StyledForm):
    equipment = forms.ModelChoiceField(label="تجهیز", queryset=Equipment.objects.none())
    project = forms.ModelChoiceField(label="پروژه", queryset=Project.objects.none())
    site = forms.ModelChoiceField(
        label="کارگاه",
        queryset=ProjectSite.objects.none(),
        required=False,
        empty_label="بدون کارگاه",
    )
    start_date = forms.DateField(
        label="تاریخ شروع", widget=forms.DateInput(attrs={"type": "date"})
    )
    end_date = forms.DateField(
        label="تاریخ پایان",
        required=False,
        widget=forms.DateInput(attrs={"type": "date"}),
    )
    purpose = forms.CharField(label="هدف تخصیص", max_length=255, required=False)
    notes = forms.CharField(
        label="یادداشت‌ها", required=False, widget=forms.Textarea(attrs={"rows": 3})
    )

    def __init__(
        self,
        *args: object,
        organization,
        membership: OrganizationMembership | None = None,
        **kwargs: object,
    ) -> None:
        super().__init__(*args, **kwargs)
        if not isinstance(membership, OrganizationMembership) or (
            membership.organization_id != organization.pk
        ):
            return
        self.fields["equipment"].queryset = Equipment.objects.filter(
            organization=organization, archived_at__isnull=True
        ).order_by("name")
        self.fields["project"].queryset = scoped_projects_for_form(
            organization=organization, membership=membership
        )
        self.fields["site"].queryset = (
            scope_queryset(
                ProjectSite.objects.filter(
                    organization=organization,
                    is_active=True,
                    project__deleted_at__isnull=True,
                ),
                membership=membership,
                project_lookup="project_id",
                site_lookup="pk",
            )
            .select_related("project")
            .order_by("project__name", "name")
        )

    def clean(self):
        cleaned = super().clean()
        site, project = cleaned.get("site"), cleaned.get("project")
        if site and project and site.project_id != project.pk:
            self.add_error("site", "کارگاه باید متعلق به پروژه انتخاب‌شده باشد.")
        return cleaned

        label="ارائه‌دهنده خدمت",

        required=False,

        label="شرح کار", required=False, widget=forms.Textarea(attrs={"rows": 3})

        label="هزینه", required=False, min_value=0, decimal_places=2

    )

        self.fields["equipment"].queryset = Equipment.objects.filter(
            organization=organization, archived_at__isnull=True

            organization=organization, deleted_at__isnull=True

        if site and (not project or site.project_id != project.pk):
            self.add_error("site", "کارگاه باید متعلق به پروژه انتخاب‌شده باشد.")
        return cleaned

class CostRateForm(_StyledForm):
    policy = forms.ModelChoiceField(
        label="سیاست پیش‌نویس/بررسی‌شده",
        queryset=EquipmentCostPolicyVersion.objects.none(),
    )
    equipment = forms.ModelChoiceField(label="تجهیز", queryset=Equipment.objects.none())
    rate_amount = forms.DecimalField(
        label="نرخ داخلی IRR", min_value=0, decimal_places=6
    )
    effective_from = forms.DateField(
        label="شروع اعتبار", widget=forms.DateInput(attrs={"type": "date"})
    )
    effective_to = forms.DateField(
        label="پایان اعتبار",
        required=False,
        widget=forms.DateInput(attrs={"type": "date"}),
    )

    def __init__(self, *args, organization, **kwargs):
        super().__init__(*args, **kwargs)
        self.fields["policy"].queryset = EquipmentCostPolicyVersion.objects.filter(
            organization=organization,
            status__in=[CostPolicyStatus.DRAFT, CostPolicyStatus.REVIEWED],
        )
        self.fields["equipment"].queryset = Equipment.objects.filter(
            organization=organization, archived_at__isnull=True
        )

    def clean(self):
        cleaned = super().clean()
        if cleaned.get("rate_amount") is not None:
            cleaned["component_snapshot"] = {"base_internal": cleaned["rate_amount"]}
        return cleaned
